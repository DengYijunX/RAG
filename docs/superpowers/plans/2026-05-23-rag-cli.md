> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 构建一个 Python CLI 工具，支持 `rag index <path>` 索引本地文档和 `rag query "<问题>"` 检索问答。

**Architecture:** 分层模块化——7 个独立模块（config/loader/splitter/embedder/store/retriever/generator）+ 1 个 CLI 入口。每个模块只暴露 1-2 个公开函数，通过明确接口串联。

**Tech Stack:** Python 3.10+, ChromaDB, langchain-text-splitters, openai SDK, PyPDF2, python-dotenv

**依赖关系:**
```
config ─────────────────────────────────────────┐
loader ──┐                                       │
         ├── splitter ──┐                        │
         │               ├── store ── retriever ──┼── cli
         │               │                        │
         └── embedder ───┘         generator ─────┘
```

```python
from dataclasses import dataclass, field
from pathlib import Path
from typing import List


@dataclass
class Document:
    content: str
    metadata: dict


SUPPORTED_SUFFIXES = {".md", ".txt", ".pdf"}


def load_file(file_path: Path) -> Document:
    suffix = file_path.suffix.lower()
    if suffix == ".pdf":
        return _load_pdf(file_path)
    else:
        return _load_text(file_path)


def load_from_dir(dir_path: str, glob_pattern: str = "*") -> List[Document]:
    path = Path(dir_path)
    if not path.is_dir():
        raise NotADirectoryError(f"{dir_path} is not a directory")

    documents = []
    for file_path in sorted(path.rglob(glob_pattern)):
        if file_path.is_file() and file_path.suffix.lower() in SUPPORTED_SUFFIXES:
            try:
                doc = load_file(file_path)
                doc.metadata["source"] = str(file_path)
                documents.append(doc)
            except Exception as e:
                print(f"Warning: failed to load {file_path}: {e}")
    return documents


def _load_text(file_path: Path) -> Document:
    with open(file_path, "r", encoding="utf-8") as f:
        content = f.read()
    return Document(content=content, metadata={"source": str(file_path)})


def _load_pdf(file_path: Path) -> Document:
    from PyPDF2 import PdfReader
    reader = PdfReader(str(file_path))
    pages = []
    for page in reader.pages:
        text = page.extract_text()
        if text:
            pages.append(text)
    content = "\n".join(pages)
    return Document(content=content, metadata={"source": str(file_path)})
```

from dataclasses import dataclass
from typing import List
from langchain_text_splitters import RecursiveCharacterTextSplitter

from rag.loader import Document


@dataclass
class Chunk:
    content: str
    metadata: dict


def split(documents: List[Document], chunk_size: int = 500, chunk_overlap: int = 50) -> List[Chunk]:
    splitter = RecursiveCharacterTextSplitter(
        chunk_size=chunk_size,
        chunk_overlap=chunk_overlap,
        separators=["\n\n", "\n", "。", ".", "？", "?", " ", ""],
    )
    chunks = []
    for doc in documents:
        texts = splitter.split_text(doc.content)
        for i, text in enumerate(texts):
            chunks.append(Chunk(
                content=text,
                metadata={**doc.metadata, "chunk_index": i}
            ))
    return chunks
```

from typing import List
from openai import OpenAI

_client = None


def _get_client(config):
    global _client
    if _client is None:
        _client = OpenAI(
            base_url=config.embedding_base_url,
            api_key=config.embedding_api_key,
        )
    return _client


def embed(texts: List[str], config) -> List[List[float]]:
    client = _get_client(config)
    response = client.embeddings.create(
        model=config.embedding_model,
        input=texts,
    )
    return [d.embedding for d in response.data]


def embed_single(text: str, config) -> List[float]:
    return embed([text], config)[0]

from typing import List
import chromadb
from chromadb.config import Settings

from rag.splitter import Chunk


_client = None
_collection = None
COLLECTION_NAME = "rag_documents"


def _get_collection(config):
    global _client, _collection
    if _collection is None:
        _client = chromadb.PersistentClient(
            path=config.persist_dir,
            settings=Settings(anonymized_telemetry=False),
        )
        _collection = _client.get_or_create_collection(name=COLLECTION_NAME)
    return _collection


def reset_collection(config):
    global _client, _collection
    _client = chromadb.PersistentClient(
        path=config.persist_dir,
        settings=Settings(anonymized_telemetry=False),
    )
    try:
        _client.delete_collection(COLLECTION_NAME)
    except Exception:
        pass
    _collection = _client.create_collection(name=COLLECTION_NAME)
    return _collection


def add(chunks: List[Chunk], vectors: List[List[float]], config) -> None:
    col = _get_collection(config)
    col.add(
        ids=[f"{c.metadata['source']}_{c.metadata['chunk_index']}" for c in chunks],
        documents=[c.content for c in chunks],
        embeddings=vectors,
        metadatas=[c.metadata for c in chunks],
    )


def search(query_vector: List[float], top_k: int, config) -> List[Chunk]:
    col = _get_collection(config)
    results = col.query(query_embeddings=[query_vector], n_results=top_k)

    chunks = []
    docs = results.get("documents")
    metas = results.get("metadatas")
    if docs and docs[0]:
        for i, doc in enumerate(docs[0]):
            meta = (metas[0][i] if metas and metas[0] else {})
            chunks.append(Chunk(content=doc, metadata=meta))
    return chunks
```

from typing import List
from openai import OpenAI
from rag.splitter import Chunk


_client = None


def _get_client(config):
    global _client
    if _client is None:
        _client = OpenAI(
            base_url=config.llm_base_url,
            api_key=config.llm_api_key,
        )
    return _client


def generate(query: str, chunks: List[Chunk], config) -> str:
    client = _get_client(config)

    context = "\n\n---\n\n".join(
        f"[来源: {c.metadata.get('source', 'unknown')}]\n{c.content}"
        for c in chunks
    )

    response = client.chat.completions.create(
        model=config.llm_model,
        messages=[
            {
                "role": "system",
                "content": "你是一个基于文档回答问题的助手。请根据提供的文档内容回答问题。如果文档中没有相关信息，请如实告知。"
            },
            {
                "role": "user",
                "content": f"参考文档：\n\n{context}\n\n问题：{query}"
            },
        ],
    )
    return response.choices[0].message.content

- [ ] **Step 2: 实现 `rag/cli.py`**

```python
import argparse
from rag.config import load_config


def main():
    config = load_config()

    parser = argparse.ArgumentParser(
        prog="rag",
        description="RAG CLI - 文档索引与检索问答",
    )
    subparsers = parser.add_subparsers(dest="command")

    # index 子命令
    index_parser = subparsers.add_parser("index", help="索引文档目录")
    index_parser.add_argument("path", help="文档目录路径")
    index_parser.add_argument("--glob", default="*", help="文件匹配模式，默认 *")

    # query 子命令
    query_parser = subparsers.add_parser("query", help="检索并回答问题")
    query_parser.add_argument("question", help="要查询的问题")
    query_parser.add_argument("--top-k", type=int, help="检索数量，默认从 .env 读取")

    args = parser.parse_args()

    if args.command == "index":
        _cmd_index(args, config)
    elif args.command == "query":
        _cmd_query(args, config)
    else:
        parser.print_help()


def _cmd_index(args, config):
    from rag.loader import load_from_dir
    from rag.splitter import split
    from rag.embedder import embed
    from rag.store import reset_collection, add

    print(f"正在索引目录: {args.path}")
    docs = load_from_dir(args.path, args.glob)
    print(f"已加载 {len(docs)} 个文档")

    if not docs:
        print("未找到支持的文档文件（.md / .txt / .pdf）")
        return

    chunks = split(docs, config.chunk_size, config.chunk_overlap)
    print(f"已切分为 {len(chunks)} 个 chunk")

    print("正在进行向量化...")
    batch_size = 20
    all_vectors = []
    for i in range(0, len(chunks), batch_size):
        batch = chunks[i:i + batch_size]
        vectors = embed([c.content for c in batch], config)
        all_vectors.extend(vectors)
        print(f"  向量化 {min(i + batch_size, len(chunks))}/{len(chunks)}")
    print("向量化完成")

    reset_collection(config)
    add(chunks, all_vectors, config)
    print(f"索引完成，共 {len(chunks)} 个 chunk 已写入向量库")


def _cmd_query(args, config):
    from rag.embedder import embed_single
    from rag.store import search
    from rag.generator import generate

    top_k = args.top_k if args.top_k is not None else config.top_k

    print(f"正在检索: {args.question}")
    query_vector = embed_single(args.question, config)
    chunks = search(query_vector, top_k, config)

    print(f"已检索到 {len(chunks)} 个相关片段，正在生成回答...\n")
    answer = generate(args.question, chunks, config)
    print(answer)
    print("\n--- 引用来源 ---")
    seen = set()
    for c in chunks:
        source = c.metadata.get("source", "unknown")
        if source not in seen:
            print(f"  - {source}")
            seen.add(source)
```
